// t: 0..255 — фаза
void rainbow(uint8_t t, uint8_t *r, uint8_t *g, uint8_t *b)
{
    uint8_t region = t / 43;       // 256/6 ≈ 43
    uint8_t remainder = (t % 43) * 6;

    uint8_t p = 0;
    uint8_t q = (255 - remainder);
    uint8_t v = 255;
    uint8_t w = remainder;

    switch(region)
    {
        case 0: *r = v; *g = w; *b = p; break;
        case 1: *r = q; *g = v; *b = p; break;
        case 2: *r = p; *g = v; *b = w; break;
        case 3: *r = p; *g = q; *b = v; break;
        case 4: *r = w; *g = p; *b = v; break;
        default:*r = v; *g = p; *b = q; break;
    }
}
